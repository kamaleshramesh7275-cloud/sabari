ESP32 Firmware Source (Arduino / C++)

Production quad-channel acquisition code 
// =====================================================================
// SENSEGuard Quad-Channel Sensor Telemetry Firmware
// Designed for ESP32 / Arduino (Baud Rate: 115200)
// Team BYTECRAFT - Ideathon 10.0 (AICF)
// =====================================================================

const int PIN_S1 = 34; // ADC1_CH6 (Sensor 1)
const int PIN_S2 = 35; // ADC1_CH7 (Sensor 2)
const int PIN_S3 = 32; // ADC1_CH4 (Sensor 3)
const int PIN_S4 = 33; // ADC1_CH5 (Sensor 4)

void setup() {
  Serial.begin(115200);
  analogReadResolution(12); // 12-bit ADC (0 - 4095)
  delay(1000);
  Serial.println("{\"status\":\"ESP32_QUAD_READY\",\"version\":\"2.2\"}");
}

float readTemperature(int pin) {
  int raw = analogRead(pin);
  if (raw < 15 || raw > 4080) return -999.0; // Open Circuit
  float voltage = (raw / 4095.0) * 3.3;
  return voltage * 100.0;
}

void loop() {
  float s1 = readTemperature(PIN_S1);
  float s2 = readTemperature(PIN_S2);
  float s3 = readTemperature(PIN_S3);
  float s4 = readTemperature(PIN_S4);

  Serial.print("{\"s1\":"); Serial.print(s1, 2);
  Serial.print(",\"s2\":"); Serial.print(s2, 2);
  Serial.print(",\"s3\":"); Serial.print(s3, 2);
  Serial.print(",\"s4\":"); Serial.print(s4, 2);
  Serial.print(",\"t\":"); Serial.print(millis());
  Serial.println("}");

  delay(100); // 10 Hz sampling rate
}
